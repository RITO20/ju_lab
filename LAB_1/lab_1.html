def number_of_elements(t):
    res = 1
    for i in t:
        res *= i
    return res

def types(d):
    types = []
    f = np.ndarray.flatten(d)
    for i in f:
        t = list(str(type(i)).split(' '))
        types.append(t[1][1:len(t[1])-2])
    return types

def freq(arr):
    g = (max(arr)-min(arr))/6
    fq = np.zeros(6)
    min_val = min(arr)
    
    for i in arr:
        if i < min_val:
            fq[0] += 1
        elif min_val <= i < min_val + g:
            fq[0] += 1  # Changed from fq[1]
        elif min_val + g <= i < min_val + (2*g):
            fq[1] += 1  # Changed from fq[2]
        elif min_val + (2*g) <= i < min_val + (3*g):
            fq[2] += 1  # Changed from fq[3]
        elif min_val + (3*g) <= i < min_val + (4*g):
            fq[3] += 1  # Changed from fq[4]
        elif min_val + (4*g) <= i < min_val + (5*g):
            fq[4] += 1  # Changed from fq[5]
        else:  # i >= min_val + (5*g)
            fq[5] += 1
    
    # Create labels correctly
    labels = [
        f"{min_val:.1f}-{min_val+g:.1f}",
        f"{min_val+g:.1f}-{min_val+2*g:.1f}",
        f"{min_val+2*g:.1f}-{min_val+3*g:.1f}",
        f"{min_val+3*g:.1f}-{min_val+4*g:.1f}",
        f"{min_val+4*g:.1f}-{min_val+5*g:.1f}",
        f"{min_val+5*g:.1f}-{max(arr):.1f}"
    ]
    
    return [fq,labels]


def normal_dist(x, mean, sd):
    prob_density = (np.pi*sd) * np.exp(-0.5*((x-mean)/sd)**2)
    return prob_density

def nd_graph(arr,mean,sd):
    nd = []
    for i in arr:
        nd.append((np.pi*sd) * np.exp(-0.5*((i-mean)/sd)**2))
    return nd
